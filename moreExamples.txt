    It should only count ones where isactive is true and deactivated is null
    WITH counts AS (
    SELECT t.`TeamID`, COUNT(DISTINCT a.`AthleteID`) AS AthleteCount
    FROM `Team` AS t
    INNER JOIN `Compliance` AS c ON c.`TeamID` = t.`TeamID` AND c.`IsActive` = 1
    INNER JOIN `Athlete` AS a ON a.`AthleteID` = c.`AthleteID` AND a.`Deactivated` IS NULL
    GROUP BY t.`TeamID`
    )
    SELECT `TeamID`, AthleteCount
    FROM counts
    WHERE AthleteCount > 0 AND AthleteCount = (SELECT MAX(AthleteCount) FROM counts)
    ORDER BY `TeamID`
    LIMIT 100
